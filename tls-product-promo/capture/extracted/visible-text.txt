[h2] 简化代码演示
[button] 运行 toy 握手
[code] // 极简演示：只展示“双方算出同一个共享秘密”的核心想法
// 真实 TLS 1.3: X25519/P-256 ECDHE + HKDF + 证书链 + Finished + AEAD
const p = 23;     // toy prime，真实 TLS 不会用这么小的数
const g = 5;      // generator

const clientPrivate = 6;
const serverPrivate = 15;

const clientPublic = modPow(g, clientPrivate, p);
const serverPublic = modPow(g, serverPrivate, p);

const clientShared = modPow(serverPublic, clientPrivate, p);
const serverShared = modPow(clientPublic, serverPrivate, p);

console.log({ clientPublic, serverPublic });
console.log(clientShared === serverShared); // true

function modPow(base, exp, mod) {
  let result = 1;
  for (let i = 0; i < exp; i += 1) {
    result = (result * base) % mod;
  }
  return result;
}
